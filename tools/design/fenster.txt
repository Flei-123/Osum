# tools/design/fenster.txt -- DREHBUCH: FENSTER, LEISTE, VOLLBILD
#
# Justins Frage, woertlich: "in Windows koennen die Fenster auch nicht
# ueber die Taskleiste geschoben werden, ich hatte dir aber gesagt das
# soll so sein, das war mein Fehler. Ist das jetzt so oder nicht?
# Bitte mach's so wie Windows."
#
# Gemessen wird NICHT, ob das Ziehen "geblockt" wird -- Windows blockt
# NICHT. Gemessen wird:
#   1. liegt die Leiste OBENAUF, auch wenn ein Fenster darunter reicht
#   2. endet MAXIMIEREN an der Arbeitsflaeche statt am Schirm
#   3. VOLLBILD (F11): ganzer Schirm, keine Leiste, keine Dekoration
#      -- und zurueck auf EXAKT die alte Geometrie
#
# F12 ist die MESSTASTE (Runde FENSTER): sie legt `wm: work` und die
# ganze Fensterliste `wm: fen` auf die Leitung, ohne etwas zu aendern.
# Ohne sie schweigt die Geometrie, weil dieser Laufer mit `nopuls`
# faehrt und der Herzschlag die Liste sonst mitbraechte.

warteauf 'launcher: ready' || 120
warte 6
# Das Startmenue kommt mit hoch -- zuerst zu, damit der Schirm frei ist.
taste esc
warte 2
taste f12
warte 2
foto 10-schreibtisch

# ------------------------------------------------- 1. ein Fenster her
# Ueber die Suche, weil der Klick auf ein angeheftetes Sinnbild von
# dessen Platz abhaengt -- die Suche nennt das Programm beim Namen.
taste meta_l
warte 3
tippe files
warte 3
klickauf lzeile0
warteauf 'explorer: ready' || 60
warte 6
taste f12
warte 2
foto 11-fenster-normal

# ------------------------------ 2. UNTER die Leiste ziehen (Windows!)
# An der Titelleiste weit nach unten. Windows LAESST das zu und legt
# das Fenster DAHINTER; die Leiste bleibt vollstaendig sichtbar.
# Gezogen wird auf die zuletzt gemeldete Titelleiste des Fensters.
ziehtitel fokus 0,600
warte 4
taste f12
warte 2
foto 12-unter-der-leiste

# ------------------------------------------------- 3. wieder herauf
ziehtitel fokus 0,-500
warte 4
taste f12
warte 2
foto 13-wieder-oben

# ------------------------------------------- 4. VOLLBILD mit F11
taste f11
warte 4
taste f12
warte 2
foto 14-vollbild

# ------------------------------------------- 5. zurueck aus Vollbild
taste f11
warte 4
taste f12
warte 2
foto 15-nach-vollbild
